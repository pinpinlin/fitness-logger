const raw = await dv.io.load(dv.current().file.path);
const cur = dv.current();
// 參數優先讀各區塊引言行（支援一場多組）；找不到才回退 frontmatter
let p = { w: cur["HIIT每項秒"], i: cur["HIIT項間休"], r: cur["HIIT輪數"], rr: cur["HIIT輪間休"] };
const groups = new Map();
for (const line of raw.split('\n')) {
  const pm = line.match(/^>\s*每項\s*(\d+)\s*秒\s*｜\s*項間休\s*(\d+)\s*秒\s*｜\s*(\d+)\s*輪\s*｜\s*輪間休\s*(\d+)/);
  if (pm) { p = { w: +pm[1], i: +pm[2], r: +pm[3], rr: +pm[4] }; continue; }
  if (!line.includes('(型態:: HIIT)')) continue;
  const nm = line.match(/\[\[([^\]|]+)(?:\|[^\]]+)?\]\]/);
  if (!nm) continue;
  const g = re => { const m = line.match(re); return m ? m[1].trim() : ''; };
  const tag = g(/\(HIIT組::\s*([^)]+)\)/) || 'A';
  if (!groups.has(tag)) groups.set(tag, { p, rows: [] });
  groups.get(tag).rows.push([dv.fileLink(nm[1].trim()), p.w ?? '', g(/\(完成::\s*([^)]+)\)/) || (p.r ?? ''), g(/\(負重::\s*([^)]+)\)/) || '徒手']);
}
if (groups.size) {
  for (const [tag, grp] of groups) {
    if (groups.size > 1) dv.header(4, `HIIT ${tag}`);
    dv.table(['動作', '每項秒', '完成輪數', '負重'], grp.rows);
    const q = grp.p;
    if (q.w && q.i != null && q.r) {
      const total = grp.rows.length * (q.w + q.i) * q.r + (q.rr ?? 0) * Math.max(0, q.r - 1);
      dv.paragraph(`設定：每項 ${q.w} 秒 ｜ 項間休 ${q.i} 秒 ｜ ${q.r} 輪 ｜ 輪間休 ${q.rr ?? 0} 秒 ｜ **總時長 ${Math.floor(total/60)}:${String(total%60).padStart(2,'0')}**`);
    }
  }
} else dv.paragraph('（本次無 HIIT）');
